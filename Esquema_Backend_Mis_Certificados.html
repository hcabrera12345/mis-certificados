Migración SQL: Función de Verificación y RLS

-- 1. Tabla de comprobantes con deduplicación de operación
CREATE TABLE public.payment_receipts (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    student_id UUID REFERENCES public.profiles(id) ON DELETE CASCADE,
    receipt_image_url TEXT NOT NULL,
    receipt_hash VARCHAR(64) UNIQUE, -- Para evitar subir la misma foto
    extracted_op_code VARCHAR(100),
    extracted_amount NUMERIC(10,2),
    ocr_status VARCHAR(30) DEFAULT 'pending',
    admin_approval_status VARCHAR(30) DEFAULT 'pending',
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. Habilitar RLS en comprobantes
ALTER TABLE public.payment_receipts ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Estudiantes insertan y ven sus propios comprobantes"
ON public.payment_receipts FOR ALL
USING (auth.uid() = student_id);

CREATE POLICY "Admins gestionan todos los comprobantes"
ON public.payment_receipts FOR ALL
USING (
    EXISTS (
        SELECT 1 FROM public.profiles
        WHERE profiles.id = auth.uid() AND profiles.role = 'admin'
    )
);
